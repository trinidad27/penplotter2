cmake_minimum_required(VERSION 3.10)
project(pantograph_plugin)
set(CMAKE_CXX_STANDARD 17)
find_package(gazebo REQUIRED)
include_directories(${GAZEBO_INCLUDE_DIRS})
link_directories(${GAZEBO_LIBRARY_DIRS})
add_library(pantograph_plugin SHARED pantograph_plugin.cc)
target_link_libraries(pantograph_plugin ${GAZEBO_LIBRARIES})
install(TARGETS pantograph_plugin LIBRARY DESTINATION ${CMAKE_INSTALL_PREFIX}/lib)
